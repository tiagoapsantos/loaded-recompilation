cmake_minimum_required(VERSION 3.16)
project(loaded_research C)
add_library(loaded_level_text STATIC src/level_text.c)
target_include_directories(loaded_level_text PUBLIC include)
set_property(TARGET loaded_level_text PROPERTY C_STANDARD 99)
add_executable(loaded-text-info src/text_info.c)
target_link_libraries(loaded-text-info PRIVATE loaded_level_text)
set_property(TARGET loaded-text-info PROPERTY C_STANDARD 99)
add_executable(loaded-text-preview src/text_preview.c)
target_link_libraries(loaded-text-preview PRIVATE loaded_level_text)
set_property(TARGET loaded-text-preview PROPERTY C_STANDARD 99)
include(CTest)
if(BUILD_TESTING)
  add_executable(test-level-text tests/test_level_text.c)
  target_link_libraries(test-level-text PRIVATE loaded_level_text)
  set_property(TARGET test-level-text PROPERTY C_STANDARD 99)
  add_test(NAME level-text-relocation COMMAND test-level-text)
endif()
add_library(loaded_level_gfx STATIC src/level_gfx.c)
target_include_directories(loaded_level_gfx PUBLIC include)
set_property(TARGET loaded_level_gfx PROPERTY C_STANDARD 99)
add_executable(loaded-gfx-info src/gfx_info.c)
target_link_libraries(loaded-gfx-info PRIVATE loaded_level_gfx)
set_property(TARGET loaded-gfx-info PROPERTY C_STANDARD 99)
if(BUILD_TESTING)
  add_executable(test-level-gfx tests/test_level_gfx.c)
  target_link_libraries(test-level-gfx PRIVATE loaded_level_gfx)
  set_property(TARGET test-level-gfx PROPERTY C_STANDARD 99)
  add_test(NAME level-gfx-offsets COMMAND test-level-gfx)
endif()
